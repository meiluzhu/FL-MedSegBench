# FeTS2022

The dataset used in this repo comes from the [Federated Tumor Segmentation (FeTS) Challenge 2022](https://www.sciencedirect.com/science/article/abs/pii/S1361841524001956).
👉 Please click [here](https://drive.google.com/drive/folders/1Elu-mGW5DPLAFfSwZcwVoJulkZd7TrUf?usp=sharing) to download.


## Dataset Description
The table below provides summary information.
For a more detailed description, please refer to the [original manuscript](https://www.sciencedirect.com/science/article/abs/pii/S1361841524001956).

|                   | Dataset description
| ----------------- | -----------------------------------------------
| Description       | This dataset is from the Federated Tumor Segmentation (FeTS) 2022 challenge.
| Dataset           | The dataset contains 1251 patients from 33 institutions. The data of each patient is acquired by multi-parametric magnetic resonance imaging (mpMRI) and has four modalities: T1-Weighted MRI (MRI-T1), postcontrast T1-weighted MRI (MRI-T1Gd), T2-Weighted MRI (MRI-T2), and T2 Fluid Attenuated Inversion Recovery (T2- FLAIR). We center-crop per modality and obtain patches with size 4 × 128 × 128 × 128 as input to the model, please the following preprocessing code.
| Centers           | We selected the institutions with more than 30 patients, forming 9 clients. 
| Task              | brain tumor segmentation. Besides normal regions, annotations comprise the GD-enhancing tumor (ET), the peritumoral edematous/invaded tissue (ED), and the necrotic tumor core (NCR).

```
# preprocess
client_names = ['site1', 'site2', 'site3', 'site10', 'site22', 'site24', 'site25', 'site26', 'site28']
save_path = data_path.replace('MICCAI_FeTS2022_TrainingData', 'MICCAI_FeTS2022_TrainingData_crop') # data_path is the path of original dataset
if not os.path.exists(save_path):
    os.makedirs(save_path)
    for client_name in client_names:
        client_id = client_name[4:]
        patients = list(df[df['Partition_ID']==client_id]['Subject'])
        
        for i, patient_id in tqdm(enumerate(patients), desc="Processing", total=len(patients)):
            patient_images=[]
            flair_path = os.path.join(data_path, patient_id, patient_id+'_flair.nii.gz')
            flair_img = sitk.ReadImage(flair_path)
            flair_img = sitk.GetArrayFromImage(flair_img) #(155, 240, 240)
            flair_img = convert_from_nii_to_png(flair_img)
            flair_img = flair_img[13:141, 56:184, 56:184]#(128, 128, 128)  
            patient_images.append(flair_img)
            
            t1_path = os.path.join(data_path, patient_id, patient_id+'_t1.nii.gz')
            t1_img = sitk.ReadImage(t1_path)
            t1_img = sitk.GetArrayFromImage(t1_img) #(155, 240, 240)
            t1_img = convert_from_nii_to_png(t1_img)
            t1_img = t1_img[13:141, 56:184, 56:184]#(128, 128, 128)  
            patient_images.append(t1_img)
            
            t1ce_path = os.path.join(data_path, patient_id, patient_id+'_t1ce.nii.gz')
            t1ce_img = sitk.ReadImage(t1ce_path)
            t1ce_img = sitk.GetArrayFromImage(t1ce_img) #(155, 240, 240)
            t1ce_img = convert_from_nii_to_png(t1ce_img)
            t1ce_img = t1ce_img[13:141, 56:184, 56:184]#(128, 128, 128)  
            patient_images.append(t1ce_img)
            
            t2_path = os.path.join(data_path, patient_id, patient_id+'_t2.nii.gz')
            t2_img = sitk.ReadImage(t2_path)
            t2_img = sitk.GetArrayFromImage(t2_img) #(155, 240, 240)
            t2_img = convert_from_nii_to_png(t2_img)
            t2_img = t2_img[13:141, 56:184, 56:184] #(128, 128, 128)  
            patient_images.append(t2_img)
            
            mask_path = os.path.join(data_path, patient_id, patient_id+'_seg.nii.gz')
            mask = sitk.ReadImage(mask_path)
            mask = sitk.GetArrayFromImage(mask) #(155, 240, 240)
            mask = mask[13:141, 56:184, 56:184] #(128, 128, 128)

            output = np.zeros_like(mask)
            output[mask == 1] = 1  # 坏死和非增强核心
            output[mask == 2] = 2  # 瘤周水肿
            output[mask == 4] = 3  # 增强肿瘤
            mask = output.astype(np.uint8)
            
            np.savez_compressed(os.path.join(save_path, patient_id+'.npz'), img_list=patient_images)
            np.savez_compressed(os.path.join(save_path, patient_id+'_seg.npz'), mask=mask)
```
